use SportsInjuryClinic

--Task 1

select *
from Patient
where Smoker = 'yes'
order by Surname

--Task 2

select * 
from Doctor
where Salary > 29999.99

--Task 3

select * 
from Appointment
where TreatmentCode = 'phy' and AppointmentDate = '2013/06/25' 
or TreatmentCode = 'mas' and AppointmentDate = '2013/06/25'

--Task 4

select Forename,Surname,AppointmentCode,AppointmentDate,
	PatientNo,InjuryCode,TreatmentCode
from Doctor D
Inner Join Appointment A
on D.StaffNo=A.StaffNo
where D.StaffNo='A1' and A.StaffNo='A1'

--Task 5

Select Forename,Surname,Salary,Salary * 1.05 as 'New_Salary',
	(salary * 1.05)-Salary as 'Increase'
from doctor

--Task 6

select AppointmentDate,D.forename,D.Surname,InjuryCode,TreatmentCode
from Appointment A
Inner join Doctor D
on A.StaffNo=D.StaffNo
where TreatmentCode='pkl' or TreatmentCode='pla'
order by AppointmentDate,D.Forename

--Task 7

select AppointmentDate,Forename,Surname,InjuryDescription,T.TreatmentCode
from Appointment A
Inner join Doctor D
on A.StaffNo=D.StaffNo
Inner Join Treatment T
On A.TreatmentCode=T.TreatmentCode
inner join injury I
On A.InjuryCode=I.InjuryCode
Where InjuryDescription like 'Broken%'

--Task 8

select AppointmentDate, COUNT (AppointmentDate) as Number_of_Patients
from Appointment
group by AppointmentDate
